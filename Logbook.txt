int	ft_parse_map_str(const char *str)// We are not using this function anymore 
{
	static int	i = 0;
	static int	width = 0;
	static int	column = 0;
	static int	height = 0;
	static int	c_count = 0;
	static int	p_count = 0;
	static int	e_count = 0; 

	if (!str || str[0] == '\0')
		return (1);
	width = ft_line_len_no_nl(str);
	if (width == 0)
		return (1);
	while (str[i])
	{
		if (str[i] == '\n')
		{
			if (column != width)
				return (1);
			height++;
			column = 0;
		}
		else
		{
			if (str[i] != '0' && str[i] != '1' && str[i] != 'C' && str[i] != 'E' && str[i] != 'P')
				return (1);
			if (str[i] == 'C')
				c_count++;
			if (str[i] == 'P')
				p_count++;
			if (str[i] == 'E')
				e_count++;
		}
		i++;
	}
	if (column > 0)
	{
		if (column != width)
			return (1);
		height++;
	}
	if (p_count != 1 || e_count != 1 || c_count < 1)
		return (1);
	return (0);	
}
